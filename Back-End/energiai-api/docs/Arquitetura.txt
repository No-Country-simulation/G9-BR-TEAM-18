br.com.energiapp
│
├── core                            {/* 1. O CORAÇÃO (Zero Spring, Zero Oracle) */}
│   ├── domain
│   │   ├── model                   # Objetos de negócio puros (Imovel, AnaliseEnergia, Equipamento)
│   │   └── exceptions              # Exceções de regra de negócio (Ex: ConsumoInvalidoException)
│   │
│   └── ports                       # As "Tomadas" (Interfaces)
│       ├── in                      # O que a aplicação FAZ (Ex: GerarAnaliseUseCase)
│       └── out                     # O que a aplicação PRECISA (Ex: AnaliseRepositoryPort, IaPythonPort)
│
├── application                     {/* 2. A ORQUESTRAÇÃO */}
│   └── services                    # Implementa as portas "in". Contém a lógica de orquestração
│                                   # (Ex: AnaliseEnergiaService implementa GerarAnaliseUseCase)
│
└── infrastructure                  {/* 3. O MUNDO EXTERNO (Depende do Spring) */}
    ├── adapters
    │   ├── in                      # Adaptadores de ENTRADA (Quem chama o nosso sistema)
    │   │   └── web
    │   │       ├── controllers     # Endpoints REST (Ex: AnaliseController)
    │   │       ├── dto             # Objetos de transferência (RequestDTO, ResponseDTO)
    │   │       └── mapper          # Converte DTO <-> Domínio
    │   │
    │   └── out                     # Adaptadores de SAÍDA (Quem o nosso sistema chama)
    │       ├── persistence
    │       │   ├── entity          # Entidades do banco com anotações JPA (@Entity, @Table)
    │       │   ├── repository      # Interfaces do Spring Data JPA
    │       │   ├── adapter         # Implementa as portas "out" (Ex: AnaliseRepositoryAdapter)
    │       │   └── mapper          # Converte Entidade JPA <-> Domínio
    │       │
    │       └── client
    │           ├── python          # Integração HTTP (FeignClient ou RestTemplate)
    │           └── adapter         # Implementa o IaPythonPort
    │
    └── config                      {/* 4. AS CONFIGURAÇÕES */}
        ├── exception               # O GlobalExceptionHandler (@ControllerAdvice)
        ├── security                # Configurações de CORS, BCrypt, Filtros
        └── BeanConfiguration.java  # Ensina o Spring a instanciar as classes do 'core'
